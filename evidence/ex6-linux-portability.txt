# Kiem tra: git clone ve VPS Linux la chay duoc ngay, khong can buoc thu cong
# Sinh lai sau khi doi default sang root + commit .gitkeep + dat HOME cho exec

## 1. Default khong can .env: ca hai container chay root nen ghi duoc bind mount
$ docker compose config | grep user:
    user: '0:0'
    user: '0:0'

## 2. Van doi uid duoc neu muon file thuoc quyen minh (.env.example)
$ AIRFLOW_UID=1000 MLFLOW_UID=1000 docker compose config | grep user:
    user: '1000:0'
    user: '1000:0'

## 3. .gitkeep DUOC commit -> thu muc bind mount ton tai ngay sau clone.
#    Neu thieu, Docker tu tao voi quyen root va container khong ghi duoc.
$ git ls-files | grep gitkeep
data/.gitkeep
logs/.gitkeep
mlflow-data/.gitkeep

## 4. HOME=/home/airflow trong compose: airflow cai bang 'pip install --user',
#    ma 'docker compose exec' khong di qua entrypoint. Thieu bien nay thi moi
#    lenh cua bai tap chet voi "No module named 'airflow'" khi uid != 50000.
$ docker compose exec airflow id -u; docker compose exec airflow airflow version
0
2.8.4
